select round(sum(tiv_2016),2) tiv_2016 from Insurance
where tiv_2015 in (
    SELECT tiv_2015 
FROM Insurance t
GROUP BY tiv_2015
HAVING COUNT(*) > 1
)
and (lat, lon) in (
    select lat, lon
    from Insurance
    group by lat, lon
    having count(*)=1
)
