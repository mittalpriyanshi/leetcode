(select u.name as results from MovieRating m
left join Users u
on m.user_id = u.user_id
group by m.user_id order by count(m.movie_id) desc, u.name asc
limit 1)

UNION ALL

(select mv.title as results  from MovieRating mo
left join Movies mv
on mo.movie_id = mv.movie_id
where mo.created_at between '2020-02-01' and  '2020-02-29'
group by mo.movie_id
order by avg(mo.rating) desc , mv.title asc
limit 1)
