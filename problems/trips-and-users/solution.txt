with cancelled as 
( select t.request_at, count(*) count_can from Trips t
join Users u1 on u1.users_id = t.client_id
join Users u2 on u2.users_id = t.driver_id
where u1.banned = 'No' and u2.banned ='No'
and status like "cancelled%"
and t.request_at between "2013-10-01" and "2013-10-03" 
group by t.request_at
)
,total as
(
    select t.request_at, count(*) as count_total from Trips t
join Users u1 on u1.users_id = t.client_id
join Users u2 on u2.users_id = t.driver_id
where u1.banned = 'No' and u2.banned ='No'
and t.request_at between "2013-10-01" and "2013-10-03" 
group by t.request_at

)
select t.request_at as 'Day', ROUND(COALESCE(c.count_can, 0) / t.count_total, 2) as 'Cancellation Rate'
from total t
left join cancelled c
on c.request_at = t.request_at
order by Day
