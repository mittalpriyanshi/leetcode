select s.student_id,s.student_name, sub.subject_name, COUNT( e.student_id) as attended_exams from Students s
cross join subjects sub
left join Examinations e  on e.student_id = s.student_id
and sub.subject_name = e.subject_name
group by e.student_id, s.student_name, sub.subject_name
order by s.student_id, sub.subject_name
