WITH ranked_table as (
    SELECT product_id, 
    year, 
    quantity, 
    price,
    RANK() OVER(PARTITION BY product_id order by year asc) as rk
    FROM Sales
)

    SELECT product_id, year as first_year, quantity, price
    FROM ranked_table
    where rk=1
    GROUP BY product_id
